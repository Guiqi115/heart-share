const shareBtn = document.getElementById('shareBtn'); const shareGuide = document.getElementById('shareGuide'); const closeGuide = document.getElementById('closeGuide'); shareBtn.addEventListener('click', async () => { // 在支持系统分享的浏览器中，优先调用系统分享 if (navigator.share) { try { await navigator.share({ title: '送你一颗会跳动的爱心', text: '点击查看这颗特别的爱心 ❤️', url: window.location.href }); return; } catch (error) { // 用户取消分享时不显示错误 if (error.name === 'AbortError') return; } } // 微信内置浏览器通常使用右上角菜单分享 shareGuide.hidden = false; }); closeGuide.addEventListener('click', () => { shareGuide.hidden = true; }); shareGuide.addEventListener('click', (event) => { if (event.target === shareGuide) { shareGuide.hidden = true; } });
